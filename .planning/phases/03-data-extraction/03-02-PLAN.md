**BNP PDF Transaction Table Format:**

```
Date de transaction | Description | Montant en euros
12 mars 2024        | AMAZON EU SARL  | -19,00
15 mars 2024        | FOREIGN STORE   | -45,00 USD
                    | Taux: 1.08      |
20 mars 2024        | NORMAL PURCHASE | -12,50
```

**Key Parsing Patterns:**

- Amount format: "-19,00" (negative = expense, comma = decimal)
- Foreign currency: "-45,00 USD" (amount + space + 3-letter currency code)
- Exchange rate on next line after foreign transaction (starts with "Taux:")
- Date column: "12 mars 2024" (handled by date-parser.ts)
- Description column: Merchant name (may contain spaces)

/**
 * Amount Parser
 *
 * Parses European number formats from BNP PDF statements.
 * BNP format: "-19,00" (comma as decimal separator)
 * Foreign currency: "-45,00 USD" (amount + currency code)
 *
 * Requirements covered:
 * - EXTRACT-03: Extract amount in EUR
 * - EXTRACT-04: Extract original currency for foreign transactions
 */

export interface ParsedAmount {
  /** Amount as number (EUR) */
  amount: number;
  /** Original currency code for foreign transactions (e.g., "USD") */
  originalCurrency?: string;
}

/**
 * Parse amount string to number
 *
 * Handles BNP format: "-19,00" → -19.00
 * Converts comma decimal separator to period
 *
 * @param amountStr - Amount string (e.g., "-19,00" or "1 234,56")
 * @returns Parsed number or null if invalid
 */
export function parseAmount(amountStr: string): number | null {
  // Normalize: trim whitespace
  const normalized = amountStr.trim();

  // Pattern: optional sign, digits with optional thousand separators, comma, 2 decimals
  // Examples: "-19,00", "1 234,56", "1234,56", "-1 234,56"
  const pattern = /^([+-]?)\s*(\d[\d\s]*)[,.](\d{2})$/;
  const match = normalized.match(pattern);

  if (!match) {
    return null;
  }

  const sign = match[1] || '';
  // Remove spaces (thousand separators)
  const wholePart = match[2].replace(/\s/g, '');
  const decimalPart = match[3];

  // Convert to number
  const value = parseFloat(`${wholePart}.${decimalPart}`);

  if (isNaN(value)) {
    return null;
  }

  // Apply sign
  return sign === '-' ? -value : value;
}

/**
 * Parse amount with optional currency code
 *
 * Handles BNP foreign transaction format: "-45,00 USD" → { amount: -45.00, originalCurrency: "USD" }
 * Also handles plain amounts: "-19,00" → { amount: -19.00 }
 *
 * @param amountStr - Amount string (e.g., "-45,00 USD" or "-19,00")
 * @returns ParsedAmount object or null if parsing fails
 */
export function parseAmountWithCurrency(amountStr: string): ParsedAmount | null {
  const normalized = amountStr.trim();

  // Pattern: amount + optional space + 3-letter currency code
  // Examples: "-45,00 USD", "-19,00", "1 234,56 GBP"
  const currencyPattern = /^(.+?)\s+([A-Z]{3})$/;
  const match = normalized.match(currencyPattern);

  if (match) {
    // Has currency code
    const amount = parseAmount(match[1]);
    const currencyCode = match[2];

    if (amount === null) {
      return null;
    }

    return {
      amount,
      originalCurrency: currencyCode,
    };
  } else {
    // No currency code, just amount
    const amount = parseAmount(normalized);

    if (amount === null) {
      return null;
    }

    return { amount };
  }
}

/**
 * Check if a line contains an exchange rate indicator
 * Exchange rate lines follow foreign currency transactions in BNP PDFs
 * Format: "Taux: 1.08" or similar
 *
 * @param line - Line of text to check
 * @returns true if this is an exchange rate line
 */
export function isExchangeRateLine(line: string): boolean {
  return /^\s*Taux:/i.test(line.trim());
}

/**
 * Extract transactions from the transaction table section
 *
 * Phase 3: Full implementation - parses individual transaction rows
 *
 * Table format:
 * Date de transaction | Description | Montant en euros
 * 12 mars 2024        | AMAZON      | -19,00
 * 15 mars 2024        | FOREIGN     | -45,00 USD
 *                     | Taux: 1.08  |
 *
 * @param tableText - Text content of the transaction table section
 * @returns Array of Transaction objects with parsed data
 */
protected extractTransactions(tableText: string): Transaction[] {
  const transactions: Transaction[] = [];
  const lines = tableText.split('\n').map(line => line.trim()).filter(line => line.length > 0);

  // Skip header line(s) - look for the actual data
  let startIndex = 0;
  for (let i = 0; i < lines.length; i++) {
    if (lines[i].includes('Date de transaction')) {
      startIndex = i + 1;
      break;
    }
  }

  // Parse transaction lines
  for (let i = startIndex; i < lines.length; i++) {
    const line = lines[i];

    // Skip exchange rate lines (they follow foreign currency transactions)
    if (isExchangeRateLine(line)) {
      continue;
    }

    // Skip footer markers
    if (this.isFooterLine(line)) {
      break;
    }

    // Try to parse as transaction
    const transaction = this.parseTransactionLine(line, lines, i);
    if (transaction) {
      transactions.push(transaction);
    }
  }

  return transactions;
}

/**
 * Check if a line is a footer marker (end of table)
 */
private isFooterLine(line: string): boolean {
  const footerMarkers = [
    'SOLDE ACTUEL',
    'TOTAL DES DEPENSES',
    'Total des dépenses',
    'Votre résultat',
    'Résultat du compte',
  ];
  return footerMarkers.some(marker => line.includes(marker));
}

/**
 * Parse a single transaction line
 *
 * Handles formats:
 * - "12 mars 2024 AMAZON EU SARL -19,00"
 * - "15 mars 2024 FOREIGN STORE -45,00 USD"
 * - "                    Taux: 1.08" (exchange rate line - skip)
 *
 * @param line - Current line
 * @param allLines - All lines for context (checking next line for exchange rate)
 * @param lineIndex - Current line index
 * @returns Transaction object or null if not a transaction line
 */
private parseTransactionLine(
  line: string,
  allLines: string[],
  lineIndex: number
): Transaction | null {
  // Pattern: date (day month year) followed by description and amount
  // Date pattern: "12 mars 2024" or "1 janvier 2024"
  const datePattern = /^(\d{1,2}\s+[a-zéû]+\s+\d{4})/i;
  const dateMatch = line.match(datePattern);

  if (!dateMatch) {
    return null; // Not a transaction line
  }

  const dateStr = dateMatch[1];
  const date = parseFrenchDate(dateStr);

  if (!date) {
    return null; // Invalid date
  }

  // Remove date from line to get description and amount
  const remainingText = line.substring(dateMatch[0].length).trim();

  // Find amount at the end of the line
  // Amount pattern: number with comma decimal, optionally followed by currency
  const amountPattern = /(.+?)\s+(-?[\d\s]+[,\.]\d{2}(?:\s+[A-Z]{3})?)$/;
  const amountMatch = remainingText.match(amountPattern);

  if (!amountMatch) {
    return null; // No amount found
  }

  const description = amountMatch[1].trim();
  const amountStr = amountMatch[2];

  const parsedAmount = parseAmountWithCurrency(amountStr);
  if (!parsedAmount) {
    return null;
  }

  return {
    date,
    description,
    amount: parsedAmount.amount,
    originalCurrency: parsedAmount.originalCurrency,
  };
}
