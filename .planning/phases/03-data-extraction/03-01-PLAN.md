/**
 * French Date Parser
 *
 * Parses French date strings from BNP PDF statements and converts to Date objects.
 * BNP format: "12 mars 2024" → Date → "3/12/2024"
 *
 * Requirements covered:
 * - EXTRACT-01: Extract transaction date (Date de transaction)
 * - EXTRACT-05: Format dates as M/D/YYYY
 */

/**
 * French month names mapping to month numbers (1-12)
 */
export const FRENCH_MONTHS: Record<string, number> = {
  janvier: 1,
  février: 2,
  mars: 3,
  avril: 4,
  mai: 5,
  juin: 6,
  juillet: 7,
  août: 8,
  septembre: 9,
  octobre: 10,
  novembre: 11,
  décembre: 12,
};

/**
 * Parse French date string to Date object
 *
 * Handles BNP format: "12 mars 2024" or "1 janvier 2024"
 * Also handles variations with extra spaces
 *
 * @param dateStr - French date string (e.g., "12 mars 2024")
 * @returns Date object or null if parsing fails
 */
export function parseFrenchDate(dateStr: string): Date | null {
  // Normalize: trim and collapse multiple spaces
  const normalized = dateStr.trim().replace(/\s+/g, ' ');

  // Pattern: day month year (e.g., "12 mars 2024")
  const pattern = /^(\d{1,2})\s+([a-zéû]+)\s+(\d{4})$/i;
  const match = normalized.match(pattern);

  if (!match) {
    return null;
  }

  const day = parseInt(match[1], 10);
  const monthName = match[2].toLowerCase();
  const year = parseInt(match[3], 10);

  const month = FRENCH_MONTHS[monthName];
  if (!month) {
    return null;
  }

  // Create date (month is 0-indexed in JavaScript Date)
  const date = new Date(year, month - 1, day);

  // Validate the date is reasonable (not in the future, not too old)
  const now = new Date();
  const minDate = new Date(now.getFullYear() - 5, 0, 1); // 5 years ago

  if (date > now || date < minDate) {
    return null; // Suspicious date range
  }

  return date;
}

/**
 * Format Date object to M/D/YYYY string for Rydoo import
 *
 * @param date - Date object
 * @returns Formatted string (e.g., "3/12/2024")
 */
export function formatDateMDY(date: Date): string {
  const month = date.getMonth() + 1; // 0-indexed to 1-indexed
  const day = date.getDate();
  const year = date.getFullYear();

  return `${month}/${day}/${year}`;
}

/**
 * Parse and format French date in one step
 * Convenience function for transaction extraction
 *
 * @param dateStr - French date string (e.g., "12 mars 2024")
 * @returns Formatted M/D/YYYY string or null if parsing fails
 */
export function parseAndFormatFrenchDate(dateStr: string): string | null {
  const date = parseFrenchDate(dateStr);
  if (!date) {
    return null;
  }
  return formatDateMDY(date);
}

import { describe, it, expect } from 'vitest';
import {
  FRENCH_MONTHS,
  parseFrenchDate,
  formatDateMDY,
  parseAndFormatFrenchDate,
} from './date-parser.js';

describe('FRENCH_MONTHS', () => {
  it('should map all 12 French month names to correct numbers', () => {
    expect(FRENCH_MONTHS['janvier']).toBe(1);
    expect(FRENCH_MONTHS['février']).toBe(2);
    expect(FRENCH_MONTHS['mars']).toBe(3);
    expect(FRENCH_MONTHS['avril']).toBe(4);
    expect(FRENCH_MONTHS['mai']).toBe(5);
    expect(FRENCH_MONTHS['juin']).toBe(6);
    expect(FRENCH_MONTHS['juillet']).toBe(7);
    expect(FRENCH_MONTHS['août']).toBe(8);
    expect(FRENCH_MONTHS['septembre']).toBe(9);
    expect(FRENCH_MONTHS['octobre']).toBe(10);
    expect(FRENCH_MONTHS['novembre']).toBe(11);
    expect(FRENCH_MONTHS['décembre']).toBe(12);
  });
});

describe('parseFrenchDate', () => {
  it('should parse "12 mars 2024" correctly', () => {
    const result = parseFrenchDate('12 mars 2024');
    expect(result).not.toBeNull();
    expect(result!.getDate()).toBe(12);
    expect(result!.getMonth()).toBe(2); // March is 2 (0-indexed)
    expect(result!.getFullYear()).toBe(2024);
  });

  it('should parse single digit days', () => {
    const result = parseFrenchDate('1 janvier 2024');
    expect(result).not.toBeNull();
    expect(result!.getDate()).toBe(1);
    expect(result!.getMonth()).toBe(0); // January
  });

  it('should parse all months', () => {
    const months = [
      ['janvier', 0],
      ['février', 1],
      ['mars', 2],
      ['avril', 3],
      ['mai', 4],
      ['juin', 5],
      ['juillet', 6],
      ['août', 7],
      ['septembre', 8],
      ['octobre', 9],
      ['novembre', 10],
      ['décembre', 11],
    ];

    for (const [monthName, expectedMonth] of months) {
      const result = parseFrenchDate(`15 ${monthName} 2024`);
      expect(result).not.toBeNull();
      expect(result!.getMonth()).toBe(expectedMonth);
    }
  });

  it('should handle extra whitespace', () => {
    const result = parseFrenchDate('  12   mars   2024  ');
    expect(result).not.toBeNull();
    expect(result!.getDate()).toBe(12);
    expect(result!.getMonth()).toBe(2);
  });

  it('should return null for invalid month', () => {
    expect(parseFrenchDate('12 invalid 2024')).toBeNull();
  });

  it('should return null for malformed dates', () => {
    expect(parseFrenchDate('not a date')).toBeNull();
    expect(parseFrenchDate('')).toBeNull();
    expect(parseFrenchDate('12/03/2024')).toBeNull(); // Wrong format
  });

  it('should return null for future dates', () => {
    const futureYear = new Date().getFullYear() + 2;
    expect(parseFrenchDate(`1 janvier ${futureYear}`)).toBeNull();
  });
});

describe('formatDateMDY', () => {
  it('should format Date to M/D/YYYY', () => {
    const date = new Date(2024, 2, 12); // March 12, 2024
    expect(formatDateMDY(date)).toBe('3/12/2024');
  });

  it('should format single digit months and days without leading zeros', () => {
    const date = new Date(2024, 0, 1); // January 1, 2024
    expect(formatDateMDY(date)).toBe('1/1/2024');
  });

  it('should format two digit months and days', () => {
    const date = new Date(2024, 11, 25); // December 25, 2024
    expect(formatDateMDY(date)).toBe('12/25/2024');
  });
});

describe('parseAndFormatFrenchDate', () => {
  it('should parse and format in one step', () => {
    expect(parseAndFormatFrenchDate('12 mars 2024')).toBe('3/12/2024');
    expect(parseAndFormatFrenchDate('1 janvier 2024')).toBe('1/1/2024');
    expect(parseAndFormatFrenchDate('25 décembre 2024')).toBe('12/25/2024');
  });

  it('should return null for invalid dates', () => {
    expect(parseAndFormatFrenchDate('invalid')).toBeNull();
  });
});
